cmake_minimum_required(VERSION 3.19)
project(AVCodecHostTests LANGUAGES CXX)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("
#ifndef __has_attribute
#error SDK availability attributes are required
#elif !__has_attribute(availability)
#error SDK availability attributes are required
#endif
int main() { return 0; }
" AVCODEC_HAS_AVAILABILITY_ATTRIBUTE)
if(NOT AVCODEC_HAS_AVAILABILITY_ATTRIBUTE)
    message(FATAL_ERROR "Select a C++ compiler with SDK availability attribute support via CXX or CMAKE_CXX_COMPILER")
endif()
enable_testing()
set(AVCODEC_TEST_TIMEOUT_SECONDS 60 CACHE STRING "Maximum time for each host test")
if(NOT AVCODEC_TEST_TIMEOUT_SECONDS MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "AVCODEC_TEST_TIMEOUT_SECONDS must be a positive integer")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OHOS_SDK_HOME "$ENV{OHOS_SDK_HOME}" CACHE PATH "OpenHarmony SDK version directory")
if(NOT EXISTS "${OHOS_SDK_HOME}/native/sysroot/usr/include")
    message(FATAL_ERROR "Set OHOS_SDK_HOME to the SDK version directory")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use a Linux host or WSL for the SDK mock tests")
endif()
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SOURCE "${ROOT}/entry/src/main/cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/targets.json" TARGETS)
string(JSON COUNT LENGTH "${TARGETS}")
math(EXPR LAST "${COUNT} - 1")
foreach(INDEX RANGE ${LAST})
    string(JSON NAME MEMBER "${TARGETS}" ${INDEX})
    add_executable(${NAME} "${NAME}.cpp")
    string(JSON SOURCE_COUNT LENGTH "${TARGETS}" ${NAME})
    if(SOURCE_COUNT GREATER 0)
        math(EXPR SOURCE_LAST "${SOURCE_COUNT} - 1")
        foreach(SOURCE_INDEX RANGE ${SOURCE_LAST})
            string(JSON FILE GET "${TARGETS}" ${NAME} ${SOURCE_INDEX})
            target_sources(${NAME} PRIVATE "${SOURCE}/${FILE}")
        endforeach()
    endif()
    target_include_directories(${NAME} PRIVATE "${SOURCE}" "${SOURCE}/common"
        "${SOURCE}/common/dfx/log" "${SOURCE}/capabilities/include")
    target_compile_options(${NAME} PRIVATE -ffunction-sections -fdata-sections -Wno-ignored-attributes
        -idirafter "${OHOS_SDK_HOME}/native/sysroot/usr/include")
    target_link_options(${NAME} PRIVATE -Wl,--gc-sections)
    target_link_libraries(${NAME} PRIVATE Threads::Threads)
    add_test(NAME ${NAME} COMMAND ${NAME})
    set_tests_properties(${NAME} PROPERTIES TIMEOUT ${AVCODEC_TEST_TIMEOUT_SECONDS})
endforeach()
